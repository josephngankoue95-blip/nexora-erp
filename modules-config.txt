    /**
     * Récupérer un module par son slug.
     */
    public static function get(string $slug): ?array
    {
        return self::all()[$slug] ?? null;
    }

    /**
     * Vérifier si un module existe.
     */
    public static function exists(string $slug): bool
    {
        return isset(self::all()[$slug]);
    }

    /**
     * Récupérer uniquement les slugs.
     */
    public static function slugs(): array
    {
        return array_keys(self::all());
    }

    /**
     * Récupérer les permissions d'un module.
     */
    public static function permissions(string $slug): array
    {
        return self::get($slug)['permissions'] ?? [];
    }

    /**
     * Récupérer les modules ayant une route.
     */
    public static function navigable(): array
    {
        return array_filter(
            self::all(),
            fn (array $module) => !empty($module['route'])
        );
    }

    /**
     * Récupérer les modules triés.
     */
    public static function sorted(): array
    {
        $modules = self::all();

        uasort(
            $modules,
            fn (array $a, array $b) =>
                $a['sort_order'] <=> $b['sort_order']
        );

        return $modules;
    }
}

<?php

namespace App\Services;

use App\Models\Company;
use App\Models\User;
use App\Support\ModuleRegistry;

class ModulePermissionService
{
    /**
     * Vérifie si un module existe dans le registre.
     */
    public function moduleExists(string $module): bool
    {
        return ModuleRegistry::exists($module);
    }

    /**
     * Récupère les permissions déclarées pour un module.
     */
    public function permissions(string $module): array
    {
        return ModuleRegistry::permissions($module);
    }

    /**
     * Vérifie si une permission appartient bien au module.
     */
    public function permissionBelongsToModule(
        string $module,
        string $permission
    ): bool {
        return in_array(
            $permission,
            $this->permissions($module),
            true
        );
    }

    /**
     * Vérifie si le module est activé pour l'entreprise.
     */
    public function moduleIsEnabled(
        ?Company $company,
        string $module
    ): bool {
        if (!$company) {
            return false;
        }

        return $company->hasModule($module);
    }

    /**
     * Vérifie l'accès complet :
     *
     * 1. Le module existe
     * 2. Le module est activé pour l'entreprise
     * 3. La permission appartient au module
     * 4. L'utilisateur possède la permission
     */
    public function can(
        User $user,
        string $module,
        string $permission
    ): bool {
        if (!$this->moduleExists($module)) {
            return false;
        }

        if (!$this->permissionBelongsToModule(
            $module,
            $permission
        )) {
            return false;
        }

        /*
        |--------------------------------------------------------------------------
        | SUPER ADMIN
        |--------------------------------------------------------------------------
        |
        | Le Super Admin n'est pas limité par les modules d'une entreprise.
        |
        */

        if ($user->isSuperAdmin()) {
            return true;
        }

        $company = $user->company;

        if (!$this->moduleIsEnabled(
            $company,
            $module
        )) {
            return false;
        }

        /*
        |--------------------------------------------------------------------------
        | PERMISSION EXISTANTE
        |--------------------------------------------------------------------------
        |
        | On conserve ton système actuel.
        |
        */

        return $user->hasPermission($permission);
    }

    /**
     * Vérifie uniquement l'accès à un module.
     */
    public function canAccessModule(
        User $user,
        string $module
    ): bool {
        if (!$this->moduleExists($module)) {
            return false;
        }

        if ($user->isSuperAdmin()) {
            return true;
        }

        return $this->moduleIsEnabled(
            $user->company,
            $module
        );
    }

    /**
     * Récupère les modules accessibles par un utilisateur.
     */
    public function accessibleModules(
        User $user
    ): array {
        if ($user->isSuperAdmin()) {
            return ModuleRegistry::sorted();
        }

        $company = $user->company;

        if (!$company) {
            return [];
        }

        $accessible = [];

        foreach (ModuleRegistry::sorted() as $slug => $module) {
            if ($this->moduleIsEnabled($company, $slug)) {
                $accessible[$slug] = $module;
            }
        }

        return $accessible;
    }
}


========== app\Services\CompanyModuleService.php ==========

<?php

namespace App\Services;

use App\Models\Company;
use Illuminate\Support\Collection;

class CompanyModuleService
{
    /**
     * Retourne les modules actifs d'une entreprise.
     */
    public function getActiveModules(?Company $company): Collection
    {
        if (!$company) {
            return collect();
        }

        return $company->activeModules()
            ->get();
    }

    /**
     * Retourne uniquement les slugs des modules actifs.
     */
    public function getActiveModuleSlugs(?Company $company): array
    {
        if (!$company) {
            return [];
        }

        return $company->activeModules()
            ->pluck('slug')
            ->toArray();
    }

    /**
     * Vérifie si un module est actif.
     */
    public function isEnabled(
        ?Company $company,
        string $slug
    ): bool {
        if (!$company) {
            return false;
        }

        return in_array(
            $slug,
            $this->getActiveModuleSlugs($company),
            true
        );
    }
}

<?php

namespace App\Models;

use Illuminate\Database\Eloquent\Factories\HasFactory;
use Illuminate\Database\Eloquent\Model;
use Illuminate\Database\Eloquent\Relations\BelongsToMany;

class Module extends Model
{
    use HasFactory;

    protected $fillable = [
        'name',
        'slug',
        'description',
        'icon',
        'is_active',
        'sort_order',
    ];

    protected $casts = [
        'is_active' => 'boolean',
    ];

    /**
     * Entreprises possédant ce module.
     */
    public function companies(): BelongsToMany
    {
        return $this->belongsToMany(
            Company::class,
            'company_modules'
        )
            ->withPivot([
                'is_active',
                'activated_at',
                'deactivated_at',
            ])
            ->withTimestamps();
    }
}

<?php

namespace App\Models;

use Illuminate\Database\Eloquent\Factories\HasFactory;
use Illuminate\Database\Eloquent\Model;
use Illuminate\Database\Eloquent\Relations\BelongsTo;

class CompanyModule extends Model
{
    use HasFactory;

    protected $table = 'company_modules';

    protected $fillable = [
        'company_id',
        'module_id',
        'is_active',
        'activated_at',
        'deactivated_at',
    ];

    protected $casts = [
        'is_active' => 'boolean',
        'activated_at' => 'datetime',
        'deactivated_at' => 'datetime',
    ];

    /**
     * Entreprise.
     */
    public function company(): BelongsTo
    {
        return $this->belongsTo(
            Company::class
        );
    }

    /**
     * Module.
     */
    public function module(): BelongsTo
    {
        return $this->belongsTo(
            Module::class
        );
    }
}
