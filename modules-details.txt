            foreach ($workspace['modules'] as $module) {

                if (
                    in_array(
                        $module,
                        $activeModules,
                        true
                    )
                ) {
                    $hasActiveModule = true;
                    break;
                }
            }

            if (!$hasActiveModule) {
                continue;
            }

            /*
            | Super Admin
            */

            if ($user->isSuperAdmin()) {
                $workspace['has_access'] = true;
                $availableWorkspaces[$slug] = $workspace;
                continue;
            }

            /*
            | Vérification des permissions.
            |
            | On utilise volontairement le système existant
            | hasPermission(), et non Laravel can().
            */

            $hasPermission = false;

            foreach ($workspace['permissions'] as $permission) {

                if ($user->hasPermission($permission)) {
                    $hasPermission = true;
                    break;
                }
            }

            if (!$hasPermission) {
                continue;
            }

            $workspace['has_access'] = true;

            $availableWorkspaces[$slug] = $workspace;
        }

        return view(
            'company.modules.index',
            compact(
                'availableWorkspaces',
                'company',
                'user'
            )
        );

<?php

namespace App\Http\Controllers\SuperAdmin;

use App\Http\Controllers\Controller;
use App\Models\Company;
use App\Models\Module;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\DB;

class ModuleController extends Controller
{
    /**
     * Afficher les modules d'une entreprise.
     */
    public function edit(Company $company)
    {
        $modules = Module::query()
            ->where('is_active', true)
            ->orderBy('sort_order')
            ->orderBy('name')
            ->get();

        $enabledModuleIds = DB::table('company_modules')
            ->where('company_id', $company->id)
            ->where('is_active', true)
            ->pluck('module_id')
            ->toArray();

        return view(
            'superadmin.companies.modules',
            compact(
                'company',
                'modules',
                'enabledModuleIds'
            )
        );
    }


    /**
     * Enregistrer les modules de l'entreprise.
     */
    public function update(
        Request $request,
        Company $company
    ) {
        $validated = $request->validate([
            'modules' => [
                'nullable',
                'array',
            ],

            'modules.*' => [
                'integer',
                'exists:modules,id',
            ],
        ]);

        $selectedModules = $validated['modules'] ?? [];

        DB::transaction(function () use (
            $company,
            $selectedModules
        ) {

            /*
            |--------------------------------------------------------------------------
            | DÉSACTIVER TOUS LES MODULES
            |--------------------------------------------------------------------------
            */

            DB::table('company_modules')
                ->where('company_id', $company->id)
                ->update([
                    'is_active' => false,
                    'deactivated_at' => now(),
                    'updated_at' => now(),
                ]);


            /*
            |--------------------------------------------------------------------------
            | ACTIVER LES MODULES SÉLECTIONNÉS
            |--------------------------------------------------------------------------
            */

            foreach ($selectedModules as $moduleId) {

                DB::table('company_modules')->updateOrInsert(
                    [
                        'company_id' => $company->id,
                        'module_id' => $moduleId,
                    ],
                    [
                        'is_active' => true,
                        'activated_at' => now(),
                        'deactivated_at' => null,
                        'updated_at' => now(),
                        'created_at' => now(),
                    ]
                );
            }
        });

        return redirect()
            ->route(
                'superadmin.companies.modules.edit',
                $company
            )
            ->with(
                'success',
                'Les modules de l’entreprise ont été mis à jour avec succès.'
            );
    }
}

<?php

namespace App\Http\Middleware;

use App\Services\ModulePermissionService;
use Closure;
use Illuminate\Http\Request;
use Symfony\Component\HttpFoundation\Response;

class EnsureModuleEnabled
{
    public function handle(
        Request $request,
        Closure $next,
        string $module
    ): Response {
        $user = $request->user();

        if (!$user) {
            abort(401);
        }

        $modulePermissionService = app(
            ModulePermissionService::class
        );

        /*
        |--------------------------------------------------------------------------
        | Super Admin
        |--------------------------------------------------------------------------
        |
        | Le Super Admin gère la plateforme et peut accéder aux modules
        | même lorsqu'ils ne sont pas activés pour une entreprise.
        |
        */

        if ($user->isSuperAdmin()) {
            return $next($request);
        }

        /*
        |--------------------------------------------------------------------------
        | Vérification du module
        |--------------------------------------------------------------------------
        */

        if (!$modulePermissionService->moduleExists($module)) {
            abort(
                404,
                'Ce module n’existe pas.'
            );
        }

        /*
        |--------------------------------------------------------------------------
        | Vérification de l'activation pour l'entreprise
        |--------------------------------------------------------------------------
        */

        if (!$modulePermissionService->canAccessModule(
            $user,
            $module
        )) {
            abort(
                403,
                'Ce module n’est pas activé pour votre entreprise.'
            );
        }

        return $next($request);
    }
}

<?php

namespace App\Http\Middleware;

use App\Services\ModulePermissionService;
use Closure;
use Illuminate\Http\Request;
use Symfony\Component\HttpFoundation\Response;

class EnsureModulePermission
{
    public function handle(
        Request $request,
        Closure $next,
        string $permission
    ): Response {
        $user = $request->user();

        if (!$user) {
            abort(401);
        }

        /*
        |--------------------------------------------------------------------------
        | Super Admin
        |--------------------------------------------------------------------------
        */

        if ($user->isSuperAdmin()) {
            return $next($request);
        }

        /*
        |--------------------------------------------------------------------------
        | Déterminer le module
        |--------------------------------------------------------------------------
        |
        | Exemple :
        |
        | purchases.view
        | purchases.create
        | purchases.cancel
        |
        | devient :
        |
        | module = purchases
        |
        */

        $module = explode('.', $permission)[0] ?? null;

        if (!$module) {
            abort(
                403,
                'Permission invalide.'
            );
        }

        $service = app(
            ModulePermissionService::class
        );

        /*
        |--------------------------------------------------------------------------
        | Vérification module + permission
        |--------------------------------------------------------------------------
        */

        if (!$service->can(
            $user,
            $module,
            $permission
        )) {
            abort(
                403,
                'Vous n’avez pas l’autorisation d’accéder à cette fonctionnalité.'
            );
        }

        return $next($request);
    }
}

<script>

document.addEventListener('DOMContentLoaded', function () {

    const form = document.getElementById(
        'company-modules-form'
    );

    const cards = document.querySelectorAll(
        '[data-module-card]'
    );

    const activateAll = document.getElementById(
        'activate-all'
    );

    const deactivateAll = document.getElementById(
        'deactivate-all'
    );


    /*
    |--------------------------------------------------------------------------
    | ACTUALISATION DES CARTES
    |--------------------------------------------------------------------------
    */

    function refreshCards() {

        cards.forEach(function (card) {

            const checkbox = card.querySelector(
                '.module-checkbox'
            );

            const statusText = card.querySelector(
                '.module-status-text'
            );


            if (checkbox.checked) {

                card.classList.add(
                    'is-active'
                );

                statusText.textContent =
                    'Activé';

            } else {

                card.classList.remove(
                    'is-active'
                );

                statusText.textContent =
                    'Désactivé';
            }

        });
    }


    /*
    |--------------------------------------------------------------------------
    | CHANGEMENT D'UN MODULE
    |--------------------------------------------------------------------------
    */

    cards.forEach(function (card) {

        const checkbox = card.querySelector(
            '.module-checkbox'
        );

        checkbox.addEventListener(
            'change',
            refreshCards
        );

    });


    /*
    |--------------------------------------------------------------------------
    | TOUT ACTIVER
    |--------------------------------------------------------------------------
    */

    activateAll.addEventListener(
        'click',
        function () {

            cards.forEach(function (card) {

                const checkbox =
                    card.querySelector(
                        '.module-checkbox'
                    );

                checkbox.checked = true;

            });

            refreshCards();

        }
    );


    /*
    |--------------------------------------------------------------------------
    | TOUT DÉSACTIVER
    |--------------------------------------------------------------------------
    */

    deactivateAll.addEventListener(
        'click',
        function () {

            cards.forEach(function (card) {

                const checkbox =
                    card.querySelector(
                        '.module-checkbox'
                    );

                checkbox.checked = false;

            });

            refreshCards();

        }
    );


    /*
    |--------------------------------------------------------------------------
    | ANTI DOUBLE SOUMISSION
    |--------------------------------------------------------------------------
    */

    form.addEventListener(
        'submit',
        function () {

            const button =
                form.querySelector(
                    'button[type="submit"]'
                );

            if (!button) {
                return;
            }

            button.disabled = true;

            button.innerHTML =
                '<span>✓</span> Enregistrement...';

        }
    );

});

</script>
